cmake_minimum_required(VERSION 3.19)

if (${TOOLCHAIN_LANGUAGE} MATCHES "MikroC")
    project(InitialTesting LANGUAGES MikroC)
else()
    project(InitialTesting LANGUAGES C ASM)
endif()

include(mikroeUtils)

# Release builds optimise for size (-Os) rather than CMake's -O3: about 400 KB
# of flash against 580 KB at -O3 and 650 KB for Debug (0.3.33, tools/xbuild),
# and the GUI port needs the room. Debug stays unoptimised (-O0).
set(CMAKE_C_FLAGS_RELEASE "-Os -DNDEBUG")
include(${CMAKE_CURRENT_SOURCE_DIR}/1ms_timer/TimerHeaderConfig.cmake)
set(LVGL_VERSION "9.4.0" CACHE STRING "" FORCE)
set(PROJECT_TYPE "lvgl_designer" CACHE STRING "" FORCE)

    driver/display_lvgl.h

    driver/display_lvgl.c
    driver/lv_port_indev.c
    1ms_Timer.h
    resource.mres
    main_screen.mscr
    main_screen.c
    main_screen.h
    motor_screen.c
    motor_screen.h
    limit.c
    limit.h
    usb_serial.h
    usb_serial.c
    tinyusb_cdc_device.c
    clicks.h
    clicks.c
    stepper3.h
    stepper3.c
    extflash.h
    extflash.c
    cal.h
    cal.c
    sysinfo.h
    sysinfo.c

    fixed.c

    link.h
